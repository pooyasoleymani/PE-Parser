cmake_minimum_required(VERSION 3.16)
project(amn_afzar_pe_parser VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(peparser src/main.cpp src/pe_parser.cpp)
target_include_directories(peparser PRIVATE src)
target_compile_options(peparser PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
option(BUILD_STATIC "Attempt a statically linked executable" OFF)
if(BUILD_STATIC AND UNIX AND NOT APPLE)
  target_link_options(peparser PRIVATE -static)
endif()
install(TARGETS peparser RUNTIME DESTINATION bin)
